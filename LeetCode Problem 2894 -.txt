SOLUTION TO LEETCODE PROBLEM 2894 - Divisible and Non-Divisible Sums Difference:

class Solution {
public:
    int differenceOfSums(int n, int m) { 
        int num1 = 0, num2 = 0; //The initial value of num 1 & num 2 is 0

        for (int  i= 1; i <= n; i++){ //Conditions for num1 & num2
            if (i%m == 0){
                num2 += i;
            } else
            num1 += i;
        }

        return (num1 - num2);
    }
};
